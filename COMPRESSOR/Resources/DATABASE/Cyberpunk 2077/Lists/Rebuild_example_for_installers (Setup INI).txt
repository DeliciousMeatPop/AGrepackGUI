::To be used in CIU Setup.ini

[CMDCommands]
setlocal EnableDelayedExpansion EnableExtensions
if exist "{app}\cp2077_snd.xtl" (
	:WEMTool
	cd /D "{app}"
	"{app}\WemTool.exe" -d


	:XToolDecode
	"{tmp}\XTool_2020\XTool" decode "{app}\cp2077_snd.xtl" "{app}\archive\pc\content\audio_1_general_unp" "{app}\archive\pc\content\audio_1_general.archive"
	"{tmp}\XTool_2020\XTool" decode "{app}\cp2077_snd.xtl" "{app}\archive\pc\content\audio_2_soundbanks_unp" "{app}\archive\pc\content\audio_2_soundbanks.archive"


	:DeleteFolders
	rd /s /q "{app}\archive\pc\content\audio_1_general_unp"
	rd /s /q "{app}\archive\pc\content\audio_2_soundbanks_unp"
)
:End
endlocal
