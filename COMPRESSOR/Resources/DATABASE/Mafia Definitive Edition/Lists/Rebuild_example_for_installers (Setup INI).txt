:XToolDecode
for /L %%i in (1, 1, !MainList.Count!) do (
	if exist "{app}\edit\audio\packages\!MainList[%%i].Name!" call "{tmp}\XTool_2020\XTool.exe" decode "{app}\mafiade.xtl" "{app}\edit\audio\packages\!MainList[%%i].Name!_extract" "{app}\edit\audio\packages\!MainList[%%i].Name!"
)
for /L %%a in (1, 1, !FolderList.Count!) do (
	if exist "!GameDir!\mafiade_!FolderList[%%a].Lang!.xtl" for /L %%i in (1, 1, !FileList.Count!) do (
		if exist "{app}\edit\audio\packages\!FolderList[%%a].Name!\!FileList[%%i].Name!" call "{tmp}\XTool_2020\XTool.exe" decode "{app}\mafiade_!FolderList[%%a].Lang!.xtl" "{app}\edit\audio\packages\!FolderList[%%a].Name!\!FileList[%%i].Name!_extract" "{app}\edit\audio\packages\!FolderList[%%a].Name!\!FileList[%%i].Name!"
	)
)
