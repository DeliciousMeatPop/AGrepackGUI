::To be used in CIU Setup.ini

[CMDCommands]
setlocal EnableDelayedExpansion EnableExtensions
if not exist "{app}\trine_snd.xtl" goto End
cd /D "{app}"
if exist "{app}\WemTool.exe" call "{app}\WemTool.exe" -d
call "{tmp}\XTool_2020\XTool.exe" decode "{app}\trine_snd.xtl" "{app}\_enchanted_edition_\audio1.fbq_extract" "{app}\_enchanted_edition_\audio1.fbq"
rmdir /s /q "{app}\_enchanted_edition_\audio1.fbq_extract"
del /F /S /Q "{app}\trine_snd.xtl"
del /F /S /Q "{app}\WemTool.exe"
TIMEOUT 3
:End
endlocal
