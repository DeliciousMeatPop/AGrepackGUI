::To be used in CIU Setup.ini

[CMDCommands]
setlocal EnableDelayedExpansion EnableExtensions
if not exist "{app}\aptr.xtl" goto End;
cd /D "{app}"
if exist "{app}\WemTool.exe" call "{app}\WemTool.exe" -d
set /a "FileList.Count=3"
set "FileList[1].Name=MUS_MEDIA.PC.PCK"
set "FileList[2].Name=MUS_MEDIA_D1.PC.PCK"
set "FileList[3].Name=VO_MEDIA.PC.PCK"
for /L %%i in (1, 1, !FileList.Count!) do (
	call "{tmp}\XTool_2020\XTool.exe" decode "{app}\aptr.xtl" "{app}\SOUNDBANKS\!FileList[%%i].Name!_extract" "{app}\SOUNDBANKS\!FileList[%%i].Name!"
	rd /s /q "{app}\SOUNDBANKS\!FileList[%%i].Name!_extract"
)		
del /F /S /Q "{app}\aptr.xtl"
del /S /F /Q "{app}\WemTool.exe
TIMEOUT 3
:End
endlocal
