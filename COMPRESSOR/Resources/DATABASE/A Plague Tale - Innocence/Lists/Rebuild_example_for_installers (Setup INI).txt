::To be used in CIU Setup.ini

[CMDCommands]
setlocal EnableDelayedExpansion EnableExtensions
if not exist "{app}\apti.xtl" goto End
cd /D "{app}"
if exist "{app}\WemTool.exe" call "{app}\WemTool.exe" -d
set /a "FileList.Count=13"
set "FileList[1].Name=MUSIC_D1_MEDIA.PC.PCK"
set "FileList[2].Name=MUSIC_MEDIA.PC.PCK"
set "FileList[3].Name=VO_AMICIA_MEDIA.PC.PCK"
set "FileList[4].Name=VO_D1_MEDIA.PC.PCK"
set "FileList[5].Name=VO_GUARD1_MEDIA.PC.PCK"
set "FileList[6].Name=VO_GUARD2_MEDIA.PC.PCK"
set "FileList[7].Name=VO_GUARD3_MEDIA.PC.PCK"
set "FileList[8].Name=VO_GUARD4_MEDIA.PC.PCK"
set "FileList[9].Name=VO_HUGO_MEDIA.PC.PCK"
set "FileList[10].Name=VO_LUCAS_MEDIA.PC.PCK"
set "FileList[11].Name=VO_MELIE_MEDIA.PC.PCK"
set "FileList[12].Name=VO_OTHERS_MEDIA.PC.PCK"
set "FileList[13].Name=VO_RODRIC_MEDIA.PC.PCK"
for /L %%i in (1, 1, !FileList.Count!) do (
	call "{tmp}\XTool_2020\XTool.exe" decode "{app}\apti.xtl" "{app}\SOUNDBANKS\!FileList[%%i].Name!_extract" "{app}\SOUNDBANKS\!FileList[%%i].Name!"
	rmdir /s /q "{app}\SOUNDBANKS\!FileList[%%i].Name!_extract"
)
del /F /S /Q "{app}\apti.xtl"
del /S /F /Q "{app}\WemTool.exe"
TIMEOUT 3
:End
endlocal
